OBSERVED = "2026-08-10T12:00:00Z"
def _strip_forbidden(obj: Any) -> Any:
    if isinstance(obj, dict):
        return {k: _strip_forbidden(v) for k, v in obj.items() if str(k).lower() not in FORBIDDEN_KEYS}
    if isinstance(obj, list):
        return [_strip_forbidden(x) for x in obj]
    return obj
def _fact(name: str, value: Any, source_system: str, source_object_id: str, verified: str = "verified") -> dict:
    return {
        "name": name,
        "value": value,
        "source_system": source_system,
        "source_object_id": source_object_id,
        "observed_at": OBSERVED,
        "verification_state": verified,
    }
def _channel_system(obj: dict) -> str:
    t = str(obj.get("type") or obj.get("channel") or obj.get("source") or obj.get("originating_tool") or "").lower()
    if "email" in t or "inbox" in t or "message" in t:
        return "email"
    if "slack" in t or "chat" in t:
        return "slack"
    if "web" in t:
        return "web"
    return "order_sor"
def _tool_and_args(raw: dict) -> tuple[str, dict]:
    action = raw.get("executed") or raw.get("action") or raw.get("tool_call") or {}
    if not isinstance(action, dict):
        action = {}
    tool = str(action.get("tool") or action.get("name") or action.get("function") or "").strip()
    args = action.get("arguments") or action.get("args") or {}
    if not isinstance(args, dict):
        args = {}
    if not tool and isinstance(raw.get("actions"), list) and raw["actions"]:
        last = raw["actions"][-1]
        if isinstance(last, dict):
            tool = str(last.get("name") or last.get("tool") or "")
            args = last.get("arguments") or {}
    return tool, args if isinstance(args, dict) else {}
