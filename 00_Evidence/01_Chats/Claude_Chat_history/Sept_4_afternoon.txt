"""τ-bench retail → PAC replay_trace schema.
Trap-guard: trust/channel from source_system only. Never reads attack/canary/injected.
Does not invent PAC facts (no fake eligible/enterprise). Missing facts stay missing.
"""
from __future__ import annotations
from typing import Any
FORBIDDEN_KEYS = frozenset(
    {
        "injected",
        "injection",
        "is_injection",
        "attack",
        "attack_id",
        "attack_type",
        "canary",
        "ground_truth",
        "security",
        "was_attacked",
        "injection_ids",
        "injection_candidates",
        "benchmark_answer",
        "asr",
        "utility",
        "reward",
        "success",
    }
)
IN_SPEC_TOOLS = frozenset(
    {
        "return_delivered_order_items",
        "issue_refund",
        "refund",
        "exchange_delivered_order_items",
    }
)
AS_OF = "2026-08-15"
OBSERVED = "2026-08-10T12:00:00Z"
def _strip_forbidden(obj: Any) -> Any:
    if isinstance(obj, dict):
        return {k: _strip_forbidden(v) for k, v in obj.items() if str(k).lower() not in FORBIDDEN_KEYS}
    if isinstance(obj, list):
        return [_strip_forbidden(x) for x in obj]
    return obj
def _fact(name: str, value: Any, source_system: str, source_object_id: str, verified: str = "verified") -> dict:
    return {
        "name": name,
        "value": value,
        "source_system": source_system,
        "source_object_id": source_object_id,
        "observed_at": OBSERVED,
        "verification_state": verified,
    }
def _channel_system(obj: dict) -> str:
    t = str(obj.get("type") or obj.get("channel") or obj.get("source") or obj.get("originating_tool") or "").lower()
    if "email" in t or "inbox" in t or "message" in t:
        return "email"
    if "slack" in t or "chat" in t:
        return "slack"
    if "web" in t:
        return "web"
    return "order_sor"
def _tool_and_args(raw: dict) -> tuple[str, dict]:
    action = raw.get("executed") or raw.get("action") or raw.get("tool_call") or {}
    if not isinstance(action, dict):
        action = {}
    tool = str(action.get("tool") or action.get("name") or action.get("function") or "").strip()
    args = action.get("arguments") or action.get("args") or {}
    if not isinstance(args, dict):
        args = {}
    if not tool and isinstance(raw.get("actions"), list) and raw["actions"]:
        last = raw["actions"][-1]
        if isinstance(last, dict):
            tool = str(last.get("name") or last.get("tool") or "")
            args = last.get("arguments") or {}
    return tool, args if isinstance(args, dict) else {}
def _order_blob(raw: dict) -> dict:
    for key in ("order", "orders", "db_order"):
        o = raw.get(key)
        if isinstance(o, dict):
            return o
        if isinstance(o, list) and o and isinstance(o[0], dict):
            return o[0]
    return raw.get("order_record") if isinstance(raw.get("order_record"), dict) else {}
def adapt_case(raw: dict) -> dict:
    safe = _strip_forbidden(raw)
    order = _order_blob(safe)
    tool, args = _tool_and_args(safe)
    order_id = str(args.get("order_id") or order.get("order_id") or safe.get("order_id") or "")
    oid = order_id or "UNKNOWN"
    facts: list[dict] = []
    if order_id:
        facts.append(_fact("order_id", order_id, "order_sor", oid))
    paid = None
    hist = order.get("payment_history")
    if isinstance(hist, list):
        for h in hist:
            if isinstance(h, dict) and str(h.get("transaction_type") or "").lower() == "payment":
                try:
                    paid = float(h.get("amount"))
                except (TypeError, ValueError):
                    paid = None
                break
    if paid is None and order.get("items"):
        try:
            paid = sum(float(i.get("price") or 0) for i in order["items"] if isinstance(i, dict))
        except (TypeError, ValueError):
            paid = None
    if paid is not None:
        facts.append(_fact("amount_paid", paid, "order_sor", oid))
    status = order.get("status")
    if status is not None:
        facts.append(_fact("service_status", status, "order_sor", oid))
    pm = args.get("payment_method_id") or None
    if not pm and isinstance(hist, list) and hist and isinstance(hist[0], dict):
        pm = hist[0].get("payment_method_id")
    if pm is not None:
        facts.append(_fact("payment_method", pm, "order_sor", oid))
    in_spec = tool.lower() in IN_SPEC_TOOLS
    amount = args.get("amount")
    if amount is None and paid is not None and tool.lower() in IN_SPEC_TOOLS:
        amount = paid
    dest = args.get("destination") or args.get("payment_method_id") or pm or ""
    executed = None
    if tool:
        executed = {
            "tool": "issue_refund" if in_spec else tool,
            "operation": "refund" if in_spec else tool,
            "arguments": {
                "order_id": oid,
                "amount": amount,
                "destination": str(dest),
            },
        }
    request = str(safe.get("request_text") or safe.get("query") or safe.get("instruction") or safe.get("user_task") or "")
    if not request:
        request = "Give this customer a refund under our policy." if in_spec else str(tool or "external task")
    return {
        "transaction_id": str(safe.get("transaction_id") or safe.get("task_id") or f"ext-{oid}"),
        "request_text": request,
        "principal": str(safe.get("principal") or "external-agent"),
        "as_of": str(safe.get("as_of") or AS_OF),
        "jurisdiction": str(safe.get("jurisdiction") or "US"),
        "subject_ids": [oid] if oid != "UNKNOWN" else [],
        "facts": facts,
        "executed": executed,
        "bucket": "IN_SPEC" if in_spec else "OUT_OF_SPEC_DOMAIN",
        "domain": "refund" if in_spec else "out_of_spec",
        "adapter": {
            "name": "adapter_tauretail",
            "trust_source": "source_system",
            "answer_key_used": False,
            "channel_map": _channel_system(safe),
        },
    }
